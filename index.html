const BOT_USERNAME = "@FVScamBot"; const TOPICS = { guarantor: 2, scam: 3, suspect: 7 }; export default { async fetch(request, env) { const url = new URL(request.url); // Проверка Worker if ( request.method === "GET" && url.pathname === "/" ) { return new Response( "FVScam Bot is online ⚜️" ); /* * Если username не указан, * используем автора сообщения. */ if ( usernames.length === 0 && message.from && message.from.username ) { usernames.push( message.from.username ); } for ( const targetUsername of usernames ) { await updateUserStatus( env, targetUsername, topicType ); } } // ================================================== // ПОИСК USERNAME // ================================================== function extractUsernames( text ) { const matches = text.match( /@[A-Za-z0-9_]{5,32}/g );
